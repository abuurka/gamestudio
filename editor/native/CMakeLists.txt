# GameStudio native module (L2 in docs/fork/ARCHITECTURE.md).
#
# Blender's build includes this directory through seam S01, at the end of
# source/blender/CMakeLists.txt, when configured with -DGS_NATIVE_DIR=<this dir>.
# Every bf_* library exists by then. Nothing here edits upstream files: we only
# set values for the hooks the seams expose.

set(GS_PRODUCT_NAME "GameStudio")
set(GS_VENDOR_DIR "GameStudio")       # Windows: %APPDATA%\GameStudio\Editor\<ver>
set(GS_APP_DIR "Editor")
set(GS_APP_DIR_MACOS "GameStudio")    # macOS:   ~/Library/Application Support/GameStudio/<ver>
set(GS_APP_DIR_UNIX "gamestudio")     # Linux:   ~/.config/gamestudio/<ver>
set(GS_DEFAULT_APP_TEMPLATE "GameStudio")

# Seam S02: GameStudio keeps its own preferences, add-ons and caches, and never reads
# or writes the user's Blender profile.
set(_gs_dirs
  BLENDER_APP_VENDOR_DIR="${GS_VENDOR_DIR}"
  BLENDER_APP_DIR="${GS_APP_DIR}"
  BLENDER_APP_DIR_MACOS="${GS_APP_DIR_MACOS}"
  BLENDER_APP_DIR_UNIX="${GS_APP_DIR_UNIX}"
)
target_compile_definitions(bf_intern_ghost PRIVATE ${_gs_dirs})
target_compile_definitions(bf_blenkernel PRIVATE ${_gs_dirs})

# Seam S03: the name shown in window titles and the About box.
# Seam S04: start in the GameStudio app template (the IDE shell) by default.
target_compile_definitions(bf_windowmanager PRIVATE
  BLENDER_APP_NAME="${GS_PRODUCT_NAME}"
  WM_DEFAULT_APP_TEMPLATE="${GS_DEFAULT_APP_TEMPLATE}"
)

# --- bf_gamestudio: extension editors and the `_gamestudio` Python module ---------------------
# Needs seams S05-S08. They add DNA_space_extension_types.h, which is how we detect them, so
# this file also works against a tree that only has the Phase 1 seams.
if(EXISTS "${CMAKE_SOURCE_DIR}/source/blender/makesdna/DNA_space_extension_types.h")
  set(INC
    ${CMAKE_SOURCE_DIR}/source/blender/editors/include
    ${CMAKE_SOURCE_DIR}/source/blender/makesrna
    # RNA_prototypes.hh
    ${CMAKE_BINARY_DIR}/source/blender/makesrna
    # GHOST_IWindow.hh: the Game View pauses the game while its window is minimized.
    ${CMAKE_SOURCE_DIR}/intern/ghost
  )
  set(INC_SYS
    ${PYTHON_INCLUDE_DIRS}
  )
  set(SRC
    core/providers.cc
    core/providers.hh
    blender/space_extension.cc
    blender/python_module.cc
    blender/game_view.cc
    blender/game_view.hh
    blender/game_view_input.cc
  )
  set(LIB
    PRIVATE bf::blenkernel
    PRIVATE bf::blenlib
    PRIVATE bf::blenloader
    PRIVATE bf::dna
    PRIVATE bf::gpu
    PRIVATE bf::intern::guardedalloc
    PRIVATE bf::windowmanager
    PRIVATE bf_editor_screen
    PRIVATE bf_editor_interface
    PRIVATE bf_rna
    ${PYTHON_LINKFLAGS}
    ${PYTHON_LIBRARIES}
  )
  blender_add_lib(bf_gamestudio "${SRC}" "${INC}" "${INC_SYS}" "${LIB}")
  # Docked Game View: frame transport from the runtime host (gameview/, docs/fork/SURFACE.md).
  add_subdirectory(gameview)
  target_link_libraries(bf_gamestudio PRIVATE gs_gameview)
  # RNA_prototypes.hh and dna_type_offsets.h are generated by makesrna/makesdna.
  add_dependencies(bf_gamestudio bf_rna)
  if(NOT GAMESTUDIO_VERSION)
    set(GAMESTUDIO_VERSION "0.0.0")
  endif()
  target_compile_definitions(bf_gamestudio PRIVATE GAMESTUDIO_VERSION_STR="${GAMESTUDIO_VERSION}")

  # Seam S06: the two call sites compile only with WITH_GS_NATIVE, and the module links itself in.
  foreach(_gs_consumer bf_editor_space_api bf_python)
    target_compile_definitions(${_gs_consumer} PRIVATE WITH_GS_NATIVE)
    target_link_libraries(${_gs_consumer} PRIVATE bf_gamestudio)
  endforeach()
  message(STATUS "GameStudio native module: extension editors enabled")
endif()

message(STATUS "GameStudio native module: ${CMAKE_CURRENT_SOURCE_DIR} (product ${GS_PRODUCT_NAME})")
